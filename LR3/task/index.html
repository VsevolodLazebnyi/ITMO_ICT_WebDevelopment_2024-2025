Описание

Сервис для аренды недвижимости:

  • Пользователь может загрузить свое объявление или откликнуться на существующие.
  • Пользователь может перейти в чат с арендодателем по объявлению
  • Пользователь может изменить свои данные
  • Пользователь может посмотреть свои сделки: посмотреть нынешнюю или прогрузить старые

class LeaseDetails(models.Model):

        ("ST", 'Short-Term'),
        ("LT", 'Long-Term'),
        ("M2M", 'Month-to-Month'),
        ("S", 'Seasonal'),
        ("SL", 'Sublease'),
        ("C", 'Commercial'),
        ("CL", 'Co-Living/Shared Space'),
        ("R2O", 'Rent-to-Own'),
    )
    lease_type = models.CharField(max_length=5, choices=LEASE_TYPE_CHOICES)
    cost = models.DecimalField(decimal_places=2, max_digits=10, blank=True, null=True)
    lease_duration = models.PositiveIntegerField(
        blank=True,
        null=True,
        help_text="Duration of the lease in months, if applicable"
    )
    additional_details = models.TextField(
        blank=True,
        help_text="Additional notes or details about the lease agreement"
    )

class PostingsManageView(ListAPIView):
    permission_classes = [IsAuthenticated]
    serializer_class = PostingDetailsSerializer

    def get_queryset(self):
        if self.request.GET.get('archive', False):
            return LeasePosting.objects.filter(real_estate__owner=self.request.user,
                                               archived=True)
        return LeasePosting.objects.filter(real_estate__owner=self.request.user)


class PostingCreateView(CreateAPIView):
    permission_classes = [IsAuthenticated]
    queryset = LeasePosting.objects.all()
    serializer_class = LeasePostingCreateSerializer

    def perform_create(self, serializer):
        house = serializer.validated_data['real_estate']
        if house.owner != self.request.user:
            from rest_framework.exceptions import PermissionDenied
            raise PermissionDenied({
                'detail': 'You do not have permission to perform this action.'
            })
        return serializer.save()


class PostingChatView(ListAPIView):
    permission_classes = [IsAuthenticated]
    serializer_class = ChatMessageListSerializer

    def get_queryset(self):
        return ChatMessage.objects.filter(
            Q(chat__posting__real_estate_id=self.kwargs['pk']) &
            Q(chat__renter=self.request.user) |
            Q(chat__posting__real_estate__owner=self.request.user)
        )

class RealEstateManagementView(ListCreateAPIView):
    permission_classes = [IsAuthenticated]

    def get_queryset(self):
        return RealEstate.objects.filter(owner=self.request.user)

    def get_serializer_class(self):
        if self.request.method == 'POST':
            return RealEstateCreateSerializer
        return RealEstateSerializer

    def perform_create(self, serializer):
        serializer.save(owner=self.request.user)
