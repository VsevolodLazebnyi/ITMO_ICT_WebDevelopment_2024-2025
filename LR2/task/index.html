Реализация простого сайта на django

Цель: овладеть практическими навыками и умениями реализации web-сервисов средствами Django 2.2.

Тема: "Список научных конференций"

Интерфейс описывает названия конференций, список тематик, место проведения, период проведения, описание конференций, описание место проведения, условия участия. Необходимо реализовать следующий функционал:

  • Регистрация новых пользователей.
  • Просмотр конференций и регистрацию авторов для выступлений.
  • Пользователь должен иметь возможность редактирования и удаления своих регистраций.
  • Написание отзывов к конференциям. При добавлении комментариев, должны сохраняться даты конференции, текст комментария, рейтинг (1-10), информация о комментаторе.
  • Администратор должен иметь возможность указания результатов выступления (рекомендован к публикации или нет) средствами Django- admin.
  • В клиентской части должна формироваться таблица, отображающая всех участников по конференциям.

from datetime import date

from django.contrib.auth.models import User
from django.core.validators import MinValueValidator, MaxValueValidator
from django.db import models


class Location(models.Model):
    address = models.CharField(max_length=250)

    def __str__(self):
        return self.address


class Topic(models.Model):
    name = models.CharField(max_length=250)
    description = models.TextField()

    def __str__(self):
        return self.name


class Speaker(models.Model):
    speaker = models.ForeignKey(User, on_delete=models.CASCADE)
    topic = models.OneToOneField(Topic, on_delete=models.CASCADE)
    approved = models.BooleanField(default=False)

    def __str__(self):
        return f'{self.speaker.username}: {self.topic.name}'


class Conference(models.Model):
    name = models.CharField(max_length=250)
    description = models.TextField()
    location = models.ForeignKey(Location, on_delete=models.CASCADE)
    dt_conference = models.DateField()
    dt_registration_deadline = models.DateField()
    speakers = models.ManyToManyField(Speaker, blank=True)

    def __str__(self):
        return self.name

    @property
    def is_past_due(self):
        return date.today() > self.dt_registration_deadline


class Registration(models.Model):
    user = models.ForeignKey(User, on_delete=models.CASCADE)
    conference = models.ForeignKey(Conference, on_delete=models.CASCADE)
    dt_registration = models.DateField(auto_now_add=True)


class Comment(models.Model):
    registration = models.OneToOneField(Registration, on_delete=models.CASCADE)
    title = models.CharField(max_length=250)
    text = models.TextField()
    rating = models.PositiveIntegerField(
        validators=[
            MinValueValidator(1),
            MaxValueValidator(10),
        ],
        help_text='Rating between 1 and 10',
    )
    dt_published = models.DateField(auto_now_add=True)

    class Meta:
        ordering = ('-rating',)


class ConferenceResult(models.Model):
    RESULT_CHOICES = (
        ('rc', 'recommended for publishing'),
        ('nr', 'not recommended for publishing'),
    )
    conference = models.OneToOneField(Conference, on_delete=models.CASCADE)
    result = models.CharField(max_length=5, choices=RESULT_CHOICES)

class ConferencesListView(ListView):
    model = Conference
    template_name = 'conferences/index.html'
    queryset = Conference.objects.filter(
        dt_registration_deadline__gt=datetime.datetime.today().date()
    ).order_by('dt_conference')

    context_object_name = 'conference_list'
    paginate_by = 4

class ConferencesArchiveListView(ListView):
    model = Conference
    template_name = 'conferences/archive.html'
    queryset = Conference.objects.filter(
        dt_registration_deadline__lte=datetime.datetime.today().date()
    ).order_by('-dt_conference')

    context_object_name = 'conference_list'
    paginate_by = 4

class ConferenceDetailView(DetailView):
    model = Conference
    template_name = 'conferences/details.html'
    queryset = Conference.objects.all()
    context_object_name = 'conference'

    def get_context_data(self, **kwargs):
        context = super(ConferenceDetailView, self).get_context_data(**kwargs)
        comments = Comment.objects.filter(
            registration__conference=context['conference'])
        is_registered = False
        is_speaker = False
        can_comment = False
        if not self.request.user.is_anonymous:
            registration = Registration.objects.filter(conference=context['conference'],
                                                       user=self.request.user)
            is_registered = registration.exists()

            is_speaker = Speaker.objects.filter(conference=context['conference'],
                                                speaker=self.request.user).exists()

            can_comment = is_registered and not Comment.objects.filter(registration=registration.first()).exists()

        context.update({
            'comments': comments,
            "ConferenceResult": ConferenceResult.objects.filter(
                conference=context['conference']).first() or False,
            "SpeakerForm": SpeakerForm,
            "IsRegistered": is_registered,
            "IsSpeaker": is_speaker,
            'CanComment': can_comment,
            "speakers": context['conference'].speakers.filter(approved=True),
            'comment_form': CommentForm,
        })
        return context

class AddCommentView(LoginRequiredMixin, View):
    template_name = 'conferences/create_timeout.html'

    def post(self, request, *args, **kwargs):
        status = ''

        try:
            registration = Registration.objects.get(conference=kwargs['pk'],
                                                    user=request.user)

            if Comment.objects.filter(registration=registration).exists():
                status['error'] = 'A comment for this registration already exists.'
                return render(request, self.template_name, {'status': status})

            form = CommentForm(request.POST)

            if form.is_valid():
                comment = form.save(commit=False)

                comment.registration = registration

                comment.save()

                status = 'Comment created successfully.'

            else:
                status = 'Form validation failed.'

        except Registration.DoesNotExist:
            status = 'Registration not found.'
        except Exception as e:
            status = str(e)

        return render(request, self.template_name, {'status': status})

@login_required
def delete_registration(request, pk):
    registration = get_object_or_404(Registration, id=pk)

    # Ensure the user owns the registration
    if registration.user != request.user:
        return HttpResponseForbidden("You are not allowed to delete this registration.")

    registration.delete()
    return redirect('/account/registrations/')

@login_required
def delete_speaker(request, pk):
    speaker = get_object_or_404(Speaker, id=pk)

    if speaker.speaker != request.user:
        return HttpResponseForbidden("You are not allowed to delete this speaker.")

    speaker.delete()
    return redirect('/account/applications/')
